def type_text(text, speed=0.05):
        for letter in text:
            print(letter, end ="", flush=True)
            time.sleep(speed)

def Combat(player, enemy):
    while True:
            type_text("===== COMBAT MODE =====")
            print(f"{player.name} vs {enemy.name}")
            print()        
            print(f"{player.name} HP: {player.health}")
            print(f"{enemy.name} HP: {enemy.health}")
            print()
            print("1. Attack")
            print("2. Inventory")
            print("3. Defend")
            print("4. Run")

            choice = input("> ")

            if choice == "1":
                enemy.health -= player.attack
                print(f"{enemy.name} took {player.attack} damage!")

                print("Player Health:", player.health)
                print("Enemy Health:", enemy.health)

                if enemy.health <= 0:
                    print("You defeated it!")
                    player.exp += 100
                    break

                else:
                    print(f"{enemy.name} Attacks!")
                    player.health -= enemy.attack
                    print(f"You took {enemy.attack} damage!")
                    print()
                    print("Player Health:", player.health)
                    print("Enemy Health:", enemy.health)

                    if player.health <= 0:
                        type_text("You Died...")
                        break

                    


                
            elif choice =="2":
                Inventory(player)
            elif choice == "3":
                if "Shield" in player.inventory:
                    print("You raised your shield!")

                    damage = enemy.attack // 2
                    player.health -= damage

                    print(f"You blocked some damage!")
                    print(f"You took {damage} damage!")

                    

                    if player.health <= 0:
                        type_text("You Died...")
                        break


                else:
                    print("You dont have a Shield!")
                    player.health -= enemy.attack

                    if player.health <= 0:
                        type_text("You Died...")
                        break

                

            elif choice == "4":
                print("You run away! and successfully escaped!")
                break
            else:
                print("Invalid Choice!")
